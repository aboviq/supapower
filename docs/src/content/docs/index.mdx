## In full

```ts
import { PGlite } from '@electric-sql/pglite';
import { createClient } from '@supabase/supabase-js';
import { supapower } from 'supapower';

const supabase = createClient('https://xyzcompany.supabase.co', 'your-publishable-key');
const pg = await PGlite.create({ extensions: { supapower } });

// Two-way sync, from here on: local reads and writes, offline included.
await pg.supapower.sync({ supabase, tables: ['todos', 'lists'] });

await pg.sql`INSERT INTO todos (title) VALUES (${'Test Supapower'})`;
```
